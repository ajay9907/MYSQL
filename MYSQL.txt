emp_id → INT

age → INT
salary → INT
department → VARCHAR(30)

CREATE DATABASE COMPANY;

CREATE TABLE EMPLOYEE(
E_ID INT PRIMARY KEY,
NAME VARCHAR(45),
AGE INT ,
SALARY INT,
DEPT VARCHAR(20)
);
