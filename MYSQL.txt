SELECT CUSTOMER.C_ID,CUSTOMER.C_NAME,CUSTOMER.CITY ,ORDERS.O_ID,ORDERS.O_DATE,ORDERS.AMOUNT FROM CUSTOMER INNER JOIN ORDERS ON CUSTOMER.C_ID=ORDERS.C_ID;

-- Find the customer name, order ID, and order amount where the order amount is greater than 50,000 --

SELECT CUSTOMER.C_NAME,ORDERS.O_ID,ORDERS.AMOUNT FROM CUSTOMER INNER JOIN ORDERS ON CUSTOMER.C_ID=ORDERS.C_ID WHERE AMOUNT>50000;

-- Display all customers, including customers who have not placed any order.
SELECT CUSTOMER.C_ID,CUSTOMER.C_NAME,CUSTOMER.CITY,ORDERS.O_ID,ORDERS.O_DATE,ORDERS.AMOUNT FROM CUSTOMER LEFT JOIN ORDERS ON CUSTOMER.C_ID=ORDERS.C_ID;

-- Display all orders, including orders that do not have a matching customer.

SELECT CUSTOMER.C_ID,CUSTOMER.C_NAME,CUSTOMER.CITY,ORDERS.O_ID,ORDERS.O_DATE,ORDERS.AMOUNT FROM CUSTOMER RIGHT JOIN ORDERS ON CUSTOMER.C_ID=ORDERS.C_ID;

-- Display the customer name, city, order ID, and order amount for customers from PUNE who have placed an order.

SELECT CUSTOMER.C_NAME , CUSTOMER.CITY ,ORDERS.O_ID ,ORDERS.AMOUNT FROM CUSTOMER INNER JOIN ORDERS ON CUSTOMER.C_ID=ORDERS.O_ID WHERE CUSTOMER.CITY='PUNE';

-- Display the customer name, order ID, and order amount for customers whose order amount is greater than ₹50,000.

SELECT CUSTOMER.C_NAME ,ORDERS.O_ID , ORDERS.AMOUNT FROM CUSTOMER INNER JOIN ORDERS ON CUSTOMER.C_ID=ORDERS.C_ID WHERE ORDERS.AMOUNT>50000;

-- Display all customers and their order details, but show only orders where the amount is greater than ₹40,000.

SELECT c.C_ID , c.C_NAME , c.CITY , o.O_ID ,o.O_DATE ,o.AMOUNT FROM CUSTOMER c LEFT JOIN ORDERS o ON c.C_ID=o.C_ID WHERE o.AMOUNT>40000;

-- Display all customers who have NOT placed any order.

SELECT c.C_ID ,c.C_NAME ,c.CITY ,o.O_ID ,o.O_DATE,o.AMOUNT FROM CUSTOMER c LEFT JOIN ORDERS o ON c.C_ID=o.C_ID WHERE o.O_ID='IS NULL';

-- Display all orders with customer name and city, but show only orders where the amount is less than ₹40,000.

SELECT c.C_ID,c.C_NAME,c.CITY,o.O_ID,o.O_DATE,o.AMOUNT FROM CUSTOMER c RIGHT JOIN ORDERS o ON c.C_ID=o.O_ID WHERE o.AMOUNT<40000;
